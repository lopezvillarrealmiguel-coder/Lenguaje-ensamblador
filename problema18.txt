.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para mostrar el resultado

.Data
    dia1 DWord 0C8H                  ;pasos del primer dia (200)
    dia2 DWord 0C8H                  ;pasos del segundo dia (200)
    dia3 DWord 0C8H                  ;pasos del tercer dia (200)
    dia4 DWord 0C8H                  ;pasos del cuarto dia (200)
    dia5 DWord 0C8H                  ;pasos del quinto dia (200)
    limite DWord 3E8H               ;meta de pasos (1000)
    bonus DWord 64H                 ;bonificacion (100)
    total DWord 00H                 ;guarda el total de pasos
    mensaje1 Byte "Meta diaria superada con bonus", 0 ;mensaje si supera la meta
    mensaje2 Byte "Objetivo basico alcanzado", 0 ;mensaje si no supera la meta
    mensaje3 Byte "Sincronizando reloj", 0 ;mensaje final
    titulo Byte "Registro de pasos", 0 ;titulo de la ventana
    titulo2 Byte "Total de pasos", 0 ;titulo del resultado

.Code
start:
    Mov Eax, dia1                   ;carga los pasos del primer dia
    Add Eax, dia2                   ;suma los pasos del segundo dia
    Add Eax, dia3                   ;suma los pasos del tercer dia
    Add Eax, dia4                   ;suma los pasos del cuarto dia
    Add Eax, dia5                   ;suma los pasos del quinto dia
    Mov total, Eax                  ;guarda el total de pasos

    Cmp Eax, limite                 ;compara el total con 1000
    Jg superar                      ;si es mayor, salta a superar

    Invoke MessageBox, 0, Addr mensaje2, Addr titulo, MB_OK ;muestra que se alcanzo el objetivo basico
    Invoke dwtoa, total, Addr carga ;convierte el total a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo2, MB_OK ;muestra el total de pasos
    Jmp salida                      ;salta al mensaje final

superar:
    Add Eax, bonus                  ;suma la bonificacion
    Mov total, Eax                  ;guarda el total con bonificacion
    Invoke MessageBox, 0, Addr mensaje1, Addr titulo, MB_OK ;muestra el mensaje de bonificacion
    Invoke dwtoa, total, Addr carga ;convierte el total a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo2, MB_OK ;muestra el total con bonificacion

salida:
    Invoke MessageBox, 0, Addr mensaje3, NULL, MB_OK ;muestra el mensaje final

End start
