.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para mostrar el total

.Data
    precio DWord 14H                ;precio de cada producto (20)
    cantidad DWord 04H              ;cantidad de productos (4)
    limite DWord 3CH                ;limite para aplicar descuento (60)
    descuento DWord 0AH             ;cantidad del descuento (10)
    total DWord 0H                  ;guarda el total de la compra
    mensaje1 Byte "No se aplica descuento", 0 ;mensaje si no alcanza el limite
    mensaje2 Byte "Descuento aplicado", 0 ;mensaje si se aplica el descuento
    mensaje3 Byte "Vuelva pronto", 0 ;mensaje de despedida
    titulo Byte "Compra", 0         ;titulo de la ventana
    titulo2 Byte "Total de compra", 0 ;titulo del total

.Code
start:
    Mov Eax, precio                 ;carga el precio
    Mov Ebx, cantidad               ;carga la cantidad
    Mul Ebx                         ;multiplica precio por cantidad
    Mov total, Eax                  ;guarda el total

    Cmp Eax, limite                 ;compara el total con 60
    Jge aplicar                     ;si es mayor o igual, aplica descuento

    Invoke MessageBox, 0, Addr mensaje1, Addr titulo, MB_OK ;muestra que no hay descuento
    Jmp mostrar                     ;salta a mostrar el total

aplicar:
    Sub Eax, descuento              ;resta el descuento
    Mov total, Eax                  ;guarda el total con descuento
    Invoke MessageBox, 0, Addr mensaje2, Addr titulo, MB_OK ;muestra que se aplico descuento

mostrar:
    Invoke dwtoa, total, Addr carga ;convierte el total a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo2, MB_OK ;muestra el total
    Invoke MessageBox, 0, Addr mensaje3, NULL, MB_OK ;muestra el mensaje final

End start
