.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para convertir el resultado

.Data
    numero1 DWord 0AH                ;(10)
    numero2 DWord 05H                ;(5)
    suma DWord 0H                    ;guarda el resultado
    titulo Byte "Suma", 0            ;titulo de la ventana

.Code
start:
    Mov Eax, numero1                 ;carga el primer numero
    Add Eax, numero2                 ;suma el segundo numero
    Mov suma, Eax                    ;guarda el resultado

    Invoke dwtoa, suma, Addr carga   ;convierte el resultado a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo, MB_OK ;muestra el resultado

End start
