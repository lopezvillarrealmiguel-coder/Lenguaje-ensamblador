.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para el resultado

.Data
    cantidadA DWord 04H              ;cantidad del producto A (4)
    precioA DWord 19H                ;precio del producto A (25)
    cantidadB DWord 02H              ;cantidad del producto B (2)
    precioB DWord 3CH                ;precio del producto B (60)
    cantidadC DWord 05H              ;cantidad del producto C (5)
    precioC DWord 0FH                ;precio del producto C (15)
    total DWord 0H                   ;guarda el costo total
    titulo Byte "Total de compra", 0 ;titulo de la ventana

.Code
start:
    Mov Eax, cantidadA               ;carga la cantidad del producto A
    Mov Ebx, precioA                 ;carga el precio del producto A
    Mul Ebx                          ;calcula el costo del producto A
    Mov total, Eax                   ;guarda el primer costo

    Mov Eax, cantidadB               ;carga la cantidad del producto B
    Mov Ebx, precioB                 ;carga el precio del producto B
    Mul Ebx                          ;calcula el costo del producto B
    Add total, Eax                   ;suma el segundo costo

    Mov Eax, cantidadC               ;carga la cantidad del producto C
    Mov Ebx, precioC                 ;carga el precio del producto C
    Mul Ebx                          ;calcula el costo del producto C
    Add total, Eax                   ;suma el tercer costo

    Invoke dwtoa, total, Addr carga  ;convierte el total a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo, MB_OK ;muestra el total

End start
