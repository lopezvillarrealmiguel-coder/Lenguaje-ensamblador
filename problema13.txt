.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para mostrar el resultado

.Data
    muestra DWord 32H               ;valor de cada muestra (50)
    cantidad DWord 04H              ;cantidad de muestras (4)
    limite DWord 0C8H               ;voltaje esperado (200)
    total DWord 0H                  ;guarda el voltaje total
    mensaje1 Byte "Voltaje optimo: balance de energia correcto", 0 ;mensaje si el voltaje es correcto
    mensaje2 Byte "Voltaje fuera de norma", 0 ;mensaje si el voltaje no coincide
    mensaje3 Byte "Registro guardado en memoria", 0 ;mensaje final
    titulo Byte "Control de voltaje", 0 ;titulo de la ventana
    titulo2 Byte "Voltaje total", 0  ;titulo del resultado

.Code
start:
    Mov Eax, muestra                ;carga el valor de la muestra
    Mov Ebx, cantidad              ;carga la cantidad de muestras
    Mul Ebx                         ;multiplica la muestra por cuatro
    Mov total, Eax                  ;guarda el voltaje total

    Cmp Eax, limite                 ;compara el total con 200
    Je correcto                     ;si son iguales, salta a correcto

    Invoke MessageBox, 0, Addr mensaje2, Addr titulo, MB_OK ;muestra que el voltaje esta fuera de norma
    Invoke dwtoa, total, Addr carga ;convierte el total a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo2, MB_OK ;muestra el voltaje total
    Jmp salida                      ;salta al mensaje final

correcto:
    Invoke MessageBox, 0, Addr mensaje1, Addr titulo, MB_OK ;muestra que el voltaje es optimo

salida:
    Invoke MessageBox, 0, Addr mensaje3, NULL, MB_OK ;muestra el mensaje final

End start
