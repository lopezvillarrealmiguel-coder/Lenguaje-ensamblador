.Data
    A DWord 05H                      ;valor de A (5)
    B DWord 04H                      ;valor de B (4)
    C DWord 03H                      ;valor de C (3)
    D DWord 02H                      ;valor de D (2)
    resultado DWord 0H               ;guarda el resultado
    titulo Byte "Operacion", 0       ;titulo de la ventana

.Code
start:
    Mov Eax, A                       ;carga el valor de A
    Mov Ebx, B                       ;carga el valor de B
    Mul Ebx                          ;multiplica A por B
    Add Eax, C                       ;suma C al resultado
    Sub Eax, D                       ;resta D al resultado
    Mov resultado, Eax               ;guarda el resultado final

    Invoke dwtoa, resultado, Addr carga ;convierte el resultado a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo, MB_OK ;muestra el resultado

End start
