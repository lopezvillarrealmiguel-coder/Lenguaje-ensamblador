.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para mostrar el resultado

.Data
    creditos1 DWord 0FH              ;creditos de la primera materia (15)
    creditos2 DWord 0FH              ;creditos de la segunda materia (15)
    creditos3 DWord 0FH              ;creditos de la tercera materia (15)
    limite DWord 2DH                 ;creditos necesarios (45)
    total DWord 00H                  ;guarda el total de creditos
    mensaje1 Byte "Habilitado para tramite de servicio social", 0 ;mensaje si alcanza los creditos
    mensaje2 Byte "Creditos pendientes para liberacion", 0 ;mensaje si faltan creditos
    mensaje3 Byte "Consulta de expediente concluida", 0 ;mensaje final
    titulo Byte "Consulta de creditos", 0 ;titulo de la ventana
    titulo2 Byte "Creditos totales", 0 ;titulo del resultado

.Code
start:
    Mov Eax, creditos1              ;carga los creditos de la primera materia
    Add Eax, creditos2              ;suma los creditos de la segunda materia
    Add Eax, creditos3              ;suma los creditos de la tercera materia
    Mov total, Eax                  ;guarda el total de creditos

    Cmp Eax, limite                 ;compara el total con 45
    Jge habilitado                  ;si es mayor o igual, salta a habilitado

    Invoke MessageBox, 0, Addr mensaje2, Addr titulo, MB_OK ;muestra que faltan creditos
    Jmp salida                      ;salta al mensaje final

habilitado:
    Invoke MessageBox, 0, Addr mensaje1, Addr titulo, MB_OK ;muestra que puede realizar el tramite

salida:
    Invoke MessageBox, 0, Addr mensaje3, NULL, MB_OK ;muestra el mensaje final

End start
