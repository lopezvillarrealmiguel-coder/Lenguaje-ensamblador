.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para mostrar el daño

.Data
    golpe1 DWord 14H                ;daño del primer golpe (20)
    golpe2 DWord 0FH                ;daño del segundo golpe (15)
    golpe3 DWord 19H                ;daño del tercer golpe (25)
    limite DWord 32H                ;limite de daño (50)
    reduccion DWord 0AH             ;reduccion de daño (10)
    total DWord 0H                  ;guarda el daño total
    mensaje1 Byte "Armadura intacta", 0 ;mensaje si el daño es menor a 50
    mensaje2 Byte "Daño recibido: armadura afectada", 0 ;mensaje si se reduce el daño
    mensaje3 Byte "Estado de combate actualizado", 0 ;mensaje final
    titulo Byte "Combate", 0        ;titulo de la ventana
    titulo2 Byte "Daño total", 0    ;titulo del daño

.Code
start:
    Mov Eax, golpe1                 ;carga el primer golpe
    Add Eax, golpe2                 ;suma el segundo golpe
    Add Eax, golpe3                 ;suma el tercer golpe
    Mov total, Eax                  ;guarda el daño total

    Cmp Eax, limite                 ;compara el daño con 50
    Jge reducir                     ;si es mayor o igual, reduce el daño

    Invoke MessageBox, 0, Addr mensaje1, Addr titulo, MB_OK ;muestra que la armadura esta intacta
    Jmp salida                      ;salta al mensaje final

reducir:
    Sub Eax, reduccion              ;resta 10 al daño total
    Mov total, Eax                  ;guarda el nuevo total
    Invoke MessageBox, 0, Addr mensaje2, Addr titulo, MB_OK ;muestra el mensaje de daño
    Invoke dwtoa, total, Addr carga ;convierte el daño a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo2, MB_OK ;muestra el daño restante

salida:
    Invoke MessageBox, 0, Addr mensaje3, NULL, MB_OK ;muestra el mensaje final

End start
