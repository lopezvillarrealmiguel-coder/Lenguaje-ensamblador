.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para mostrar los resultados

.Data
    base DWord 0AH                   ;base del rectangulo (10)
    altura DWord 05H                 ;altura del rectangulo (5)
    dos DWord 02H                    ;numero utilizado para el perimetro (2)
    perimetro DWord 0H               ;guarda el perimetro
    area DWord 0H                    ;guarda el area
    titulo Byte "Rectangulo", 0      ;titulo de la ventana
    titulo2 Byte "Perimetro", 0      ;titulo del perimetro
    titulo3 Byte "Area", 0           ;titulo del area

.Code
start:
    Mov Eax, base                    ;carga la base
    Add Eax, altura                 ;suma la altura
    Mul dos                         ;multiplica la suma por 2
    Mov perimetro, Eax              ;guarda el perimetro

    Mov Eax, base                    ;carga la base
    Mov Ebx, altura                 ;carga la altura
    Mul Ebx                          ;multiplica base por altura
    Mov area, Eax                   ;guarda el area

    Invoke dwtoa, perimetro, Addr carga ;convierte el perimetro a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo2, MB_OK ;muestra el perimetro

    Invoke dwtoa, area, Addr carga  ;convierte el area a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo3, MB_OK ;muestra el area

End start
