.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para mostrar el total

.Data
    peso1 DWord 46H                 ;peso de la primera persona (70)
    peso2 DWord 41H                 ;peso de la segunda persona (65)
    peso3 DWord 4BH                 ;peso de la tercera persona (75)
    limite DWord 0D2H               ;peso esperado (210)
    total DWord 0H                  ;guarda la suma de los pesos
    mensaje1 Byte "Carga de calibracion exacta detectada", 0 ;mensaje si el total es 210
    mensaje2 Byte "Carga estandar de operacion", 0 ;mensaje si el total no es 210
    mensaje3 Byte "Puertas cerrandose", 0 ;mensaje final
    titulo Byte "Control de carga", 0 ;titulo de la ventana
    titulo2 Byte "Peso total", 0    ;titulo del peso total

.Code
start:
    Mov Eax, peso1                  ;carga el primer peso
    Add Eax, peso2                  ;suma el segundo peso
    Add Eax, peso3                  ;suma el tercer peso
    Mov total, Eax                  ;guarda el peso total

    Cmp Eax, limite                 ;compara el total con 210
    Je exacto                       ;si son iguales, salta a exacto

    Invoke MessageBox, 0, Addr mensaje2, Addr titulo, MB_OK ;muestra el mensaje de carga estandar
    Invoke dwtoa, total, Addr carga ;convierte el peso total a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo2, MB_OK ;muestra el peso total
    Jmp salida                      ;salta al mensaje final

exacto:
    Invoke MessageBox, 0, Addr mensaje1, Addr titulo, MB_OK ;muestra el mensaje de carga exacta

salida:
    Invoke MessageBox, 0, Addr mensaje3, NULL, MB_OK ;muestra el mensaje final

End start
