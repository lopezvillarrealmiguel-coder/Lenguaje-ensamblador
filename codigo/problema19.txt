.Const

.Data?
    carga Byte 10 Dup(?)             ;reserva espacio para mostrar el resultado

.Data
    gasto_sector DWord 0AH          ;consumo por sector (10)
    cantidad DWord 03H              ;cantidad de sectores (3)
    limite DWord 1EH                ;limite de consumo (30)
    reduccion DWord 0AH             ;reduccion de potencia (10)
    total DWord 00H                 ;guarda el consumo total
    mensaje1 Byte "Consumo eficiente: modo crucero activo", 0 ;mensaje si el consumo es eficiente
    mensaje2 Byte "Consumo excesivo: reduciendo potencia de impulso", 0 ;mensaje si supera el limite
    mensaje3 Byte "Telemetria transmitida", 0 ;mensaje final
    titulo Byte "Consumo de combustible", 0 ;titulo de la ventana
    titulo2 Byte "Consumo total", 0  ;titulo del resultado

.Code
start:
    Mov Eax, gasto_sector           ;carga el consumo por sector
    Mov Ebx, cantidad               ;carga la cantidad de sectores
    Mul Ebx                         ;multiplica el consumo por tres
    Mov total, Eax                  ;guarda el consumo total

    Cmp Eax, limite                 ;compara el consumo con 30
    Jg excesivo                     ;si es mayor, salta a excesivo

    Invoke MessageBox, 0, Addr mensaje1, Addr titulo, MB_OK ;muestra que el consumo es eficiente
    Jmp salida                      ;salta al mensaje final

excesivo:
    Invoke MessageBox, 0, Addr mensaje2, Addr titulo, MB_OK ;muestra el mensaje de consumo excesivo
    Sub Eax, reduccion              ;resta 10 al consumo total
    Mov total, Eax                  ;guarda el consumo corregido
    Invoke dwtoa, total, Addr carga ;convierte el total a texto
    Invoke MessageBox, 0, Addr carga, Addr titulo2, MB_OK ;muestra el total corregido

salida:
    Invoke MessageBox, 0, Addr mensaje3, NULL, MB_OK ;muestra el mensaje final

End start
